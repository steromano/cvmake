% Custom commands
\newcommand{\cvheader}[5]{
    \begin{center}
        {\Huge\bfseries\color{darkgray} #1} \\[5pt]
        {\large\color{accentcolor} #2} \\[8pt]
        {\small
        \faEnvelope\ \href{mailto:#3}{#3} \quad
        \faPhone\ #4 \quad
        \faMapMarker\ #5}
    \end{center}
    \vspace{5pt}
}

\newcommand{\cvlinks}[4]{
    \begin{center}
        {\small
        \ifx&#1&\else\faLinkedin\ \href{#1}{LinkedIn} \quad\fi
        \ifx&#2&\else\faGithub\ \href{#2}{GitHub} \quad\fi
        \ifx&#3&\else\faGlobe\ \href{#3}{Website} \quad\fi
        #4}
    \end{center}
    \vspace{5pt}
}

\newcommand{\cventry}[5]{
    \textbf{#1} \hfill {\small\color{darkgray} #2} \\
    {\color{accentcolor}\textit{#3}} \hfill {\small\color{darkgray} #4} \\
    #5
    \vspace{8pt}
}

\newcommand{\cveducation}[5]{
    \textbf{#1} \hfill {\small\color{darkgray} #2} \\
    {\textit{#3}} \hfill {\small\color{darkgray} #4} \\
    #5
    \vspace{8pt}
}

\newcommand{\cvproject}[3]{
    \textbf{#1} \hfill {\small\color{darkgray} #2} \\
    #3
    \vspace{6pt}
}
