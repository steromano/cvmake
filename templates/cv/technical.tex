% Custom commands
\newcommand{\cvheader}[5]{
    \begin{center}
        {\LARGE\bfseries\color{codegray} #1} \\[3pt]
        {\normalsize\color{techblue} #2} \\[6pt]
        {\small
        \faEnvelope\ \href{mailto:#3}{#3} \quad
        \faPhone\ #4 \quad
        \faMapMarker\ #5}
    \end{center}
    \vspace{3pt}
}

\newcommand{\cvlinks}[4]{
    \begin{center}
        {\small
        \ifx&#1&\else\faLinkedin\ \href{#1}{#1} \quad\fi
        \ifx&#2&\else\faGithub\ \href{#2}{#2} \quad\fi
        \ifx&#3&\else\faGlobe\ \href{#3}{#3} \quad\fi
        #4}
    \end{center}
    \vspace{3pt}
}

\newcommand{\cventry}[5]{
    \textbf{#1} \hfill {\small\color{codegray} #2} \\
    {\color{techblue}\textit{#3}} \hfill {\small\color{codegray} #4} \\
    #5
    \vspace{6pt}
}

\newcommand{\cvproject}[4]{
    \textbf{#1} \hfill {\small\color{codegray} #2} \\
    {\small\textit{Tech: #3}} \\
    #4
    \vspace{4pt}
}

\newcommand{\skillgroup}[2]{
    \textbf{#1:} #2
    \vspace{2pt}
}
