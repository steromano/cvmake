% Custom commands
\newcommand{\cvheader}[5]{
    \begin{center}
        {\LARGE\bfseries #1} \\[6pt]
        {\normalsize #2} \\[8pt]
        #3 $\vert$ #4 $\vert$ #5
    \end{center}
    \vspace{8pt}
}

\newcommand{\cvlinks}[3]{
    \begin{center}
        \ifx&#1&\else LinkedIn: \href{#1}{#1} \quad\fi
        \ifx&#2&\else GitHub: \href{#2}{#2} \quad\fi
        \ifx&#3&\else #3\fi
    \end{center}
    \vspace{5pt}
}

\newcommand{\cventry}[5]{
    \noindent
    \textbf{#1} \hfill #2 \\
    \textit{#3} \hfill #4 \\[4pt]
    #5
    \vspace{10pt}
}

\newcommand{\cveducation}[5]{
    \noindent
    \textbf{#1} \hfill #2 \\
    #3 \hfill #4 \\[4pt]
    #5
    \vspace{10pt}
}
